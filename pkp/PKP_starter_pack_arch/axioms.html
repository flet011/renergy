// === Utility ===

function polarToCartesian(cx: number, cy: number, r: number, angleDeg: number) {

  const angleRad = (angleDeg - 90) * (Math.PI / 180);

  return {

    x: cx + r * Math.cos(angleRad),

    y: cy + r * Math.sin(angleRad),

  };

}

export default function PsychologicalAtlas() {

  const svgRef = useRef<SVGSVGElement | null>(null);

  const [selectedId, setSelectedId] = useState<string | null>(null);

  const [query, setQuery] = useState("");

  const [showSchools, setShowSchools] = useState(true);

  const [showShadow, setShowShadow] = useState(true);

  const [showAllEdges, setShowAllEdges] = useState(false);

      const rect = el.getBoundingClientRect();

      setDimensions({ width: Math.max(700, rect.width), height: Math.max(500, rect.height) });

    });

    ro.observe(el);

    return () => ro.disconnect();

  }, []);


  const { nodes, edges } = useMemo(() => {

    const W = dimensions.width;

    const H = dimensions.height;

    const cx = W / 2;

    const cy = H / 2;


    const radiusAxioms = Math.min(W, H) * 0.28;

    const radiusSchools = Math.min(W, H) * 0.42;

    const radiusShadow = Math.min(W, H) * 0.14;


    const axiomNodes = AXIOMS.map((a, i) => {

      const angle = (360 / AXIOMS.length) * i;

      const p = polarToCartesian(cx, cy, radiusAxioms, angle);

      return { id: a.id, label: a.label, type: "axiom", x: p.x, y: p.y };

    });


    const schoolNodes = SCHOOLS.map((s, i) => {

      const angle = (360 / SCHOOLS.length) * i;

      const p = polarToCartesian(cx, cy, radiusSchools, angle);

    });

      const angle = (360 / SHADOWS.length) * i;

    const nodes = [

      ...axiomNodes,

      ...(showSchools ? schoolNodes : []),

      ...(showShadow ? shadowNodes : []),

    ];

    if (showSchools) {

      for (const a of AXIOMS) {

        for (const s of SCHOOLS) {

          edges.push({ source: a.id, target: s.id, type: "light" });

        }

      }

    }

    if (showShadow) {

      for (const a of AXIOMS) {

        for (const d of SHADOWS) {

          edges.push({ source: a.id, target: d.id, type: "shadow" });

        }

      }

    }

  }, [dimensions, showSchools, showShadow]);

    if (!query.trim()) return nodes;

    const q = query.toLowerCase();

    return nodes.filter((n) => n.label.toLowerCase().includes(q));

  }, [nodes, query]);

  const visibleEdges = useMemo(() => {

    if (!showAllEdges && selectedId) {

    }

    if (!showAllEdges && !selectedId) {

      return [];

    }

    return edges.filter((e) => filteredNodeIds.has(e.source) && filteredNodeIds.has(e.target));

  }, [edges, selectedId, showAllEdges, filteredNodeIds]);


  const selectedNode = useMemo(() => nodes.find((n) => n.id === selectedId) || null, [nodes, selectedId]);


  const panelContent = useMemo(() => {

              ))}

            </div>

          </div>

          <div>

            <h4 className="font-medium mb-2">Dark Triad Distortions</h4>

            <div className="grid grid-cols-1 gap-2">

              {SHADOWS.map((d) => (

                <div key={d.id} className="text-sm"><span className="font-semibold">{d.label}:</span> {shadow[d.id] || "—"}</div>

              ))}

            </div>

          </div>

        </div>

      );

    }

    if (selectedNode.type === "school") {

      const descriptions: Record<string, string> = {

        S1: "Depth psychology: unconscious drives, conflict, meaning.",

        S2: "Learning via conditioning; environment → behavior.",

        S3: "Human potential, agency, authenticity, growth.",

        S5: "Biopsych: brain, hormones, genetics, physiology.",

        S6: "Adaptation across time; fitness of strategies.",

        <div className="space-y-3">

      );

    if (selectedNode.type === "shadow") {

          <p className="text-sm">{desc[selectedNode.id] || ""}</p>

        </div>

      );

    }

    return null;

  }, [selectedNode]);


  return (

    <div className="w-full h-full grid grid-cols-1 xl:grid-cols-12 gap-4 p-4">

              <Button onClick={() => setShowAllEdges((v) => !v)}>{showAllEdges ? "Hide Edges" : "Show Edges"}</Button>

            </div>


            {selectedNode && (

              <div className="pt-2">

                <div className="flex items-center justify-between mb-2">

            {!selectedNode && (

              </div>

            )}

          </CardContent>

        </Card>

      </div>


      {/* Graph */}

      <div className="xl:col-span-9">

        <Card className="w-full h-[70vh] xl:h-[85vh] shadow-sm">

                    y1={s.y}

                    x2={t.x}

                    y2={t.y}

                    stroke={color}

                    strokeWidth={1}

                    opacity={opacity}

                  />

                );

              })}


              {/* Nodes */}

              {nodes.map((n) => {

                const isFilteredOut = query && !n.label.toLowerCase().includes(query.toLowerCase());

                const isSelected = selectedId === n.id;

                const r = n.type === "axiom" ? 18 : n.type === "school" ? 14 : 12;

                const fill = n.type === "axiom" ? "#0ea5e9" : n.type === "school" ? "#10b981" : "#ef4444"; // cyan, green, red

                const stroke = isSelected ? "#111827" : "#334155";

                const opacity = isFilteredOut ? 0.25 : 1;

                return (

                  <g key={n.id} transform={`translate(${n.x},${n.y})`} className="cursor-pointer" onClick={() => setSelectedId(n.id)}>

                    <circle r={r} fill={fill} stroke={stroke} strokeWidth={isSelected ? 3 : 1.5} opacity={opacity} />

                    <text

                      x={0}

                      y={r + 14}

                      textAnchor="middle"

                      className="fill-slate-700 text-[10px] select-none"

                    >

                      {n.label}

                    </text>

                    {n.type !== "axiom" && (

                      <text x={0} y={r + 26} textAnchor="middle" className="fill-slate-400 text-[9px] select-none">

                        {n.type}

                      </text>

                    )}

                  </g>

                );

              })}


              {/* Concentric guides */}

              <circle cx={dimensions.width/2} cy={dimensions.height/2} r={Math.min(dimensions.width, dimensions.height)*0.14} fill="none" stroke="#e2e8f0" strokeDasharray="4 6" />

              <circle cx={dimensions.width/2} cy={dimensions.height/2} r={Math.min(dimensions.width, dimensions.height)*0.28} fill="none" stroke="#e2e8f0" strokeDasharray="4 6" />

              <circle cx={dimensions.width/2} cy={dimensions.height/2} r={Math.min(dimensions.width, dimensions.height)*0.42} fill="none" stroke="#e2e8f0" strokeDasharray="4 6" />


              {/* Legend */}

              <g transform="translate(20,30)">

                <rect x={-10} y={-18} width={220} height={70} rx={10} fill="white" stroke="#e2e8f0" />

                <g>

                  <circle cx={0} cy={0} r={6} fill="#0ea5e9" />

                  <text x={14} y={4} className="fill-slate-600 text-[11px]">Axiom</text>

                </g>

                <g transform="translate(90,0)">

                  <circle cx={0} cy={0} r={6} fill="#10b981" />

                  <text x={14} y={4} className="fill-slate-600 text-[11px]">School</text>

                </g>

                <g transform="translate(170,0)">

                  <circle cx={0} cy={0} r={6} fill="#ef4444" />

                  <text x={14} y={4} className="fill-slate-600 text-[11px]">Dark Triad</text>

                </g>

                <g transform="translate(0,24)">

                  <line x1={-2} y1={0} x2={18} y2={0} stroke="#94a3b8" strokeWidth={2} />

                  <text x={24} y={4} className="fill-slate-600 text-[11px]">Light-edge</text>

                </g>

                <g transform="translate(90,24)">

                  <line x1={-2} y1={0} x2={18} y2={0} stroke="#ef4444" strokeWidth={2} />

                  <text x={24} y={4} className="fill-slate-600 text-[11px]">Shadow-edge</text>
